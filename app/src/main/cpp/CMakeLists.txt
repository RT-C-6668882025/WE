cmake_minimum_required(VERSION 3.22.1)
project(we_drawpile C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(WE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(DRAWPILE_SOURCE_ROOT "${WE_ROOT}/vendor/drawpile" CACHE PATH "Pinned Drawpile sources")
set(ZSTD_SOURCE_ROOT "${WE_ROOT}/vendor/zstd" CACHE PATH "Pinned Zstandard sources")
set(D "${DRAWPILE_SOURCE_ROOT}/src/drawdance")
set(Z "${ZSTD_SOURCE_ROOT}/lib")
if(NOT EXISTS "${D}/libengine/dpengine/brush_engine.c" OR NOT EXISTS "${Z}/zstd.h")
  message(FATAL_ERROR "Run bash tools/fetch-native-sources.sh before building WE")
endif()
add_compile_definitions(_GNU_SOURCE DP_PROJECT_DIR_LENGTH=0 DP_BYTE_ORDER=1 DP_PERF_REALM=we_drawdance ZSTD_DISABLE_ASM)
file(GLOB common CONFIGURE_DEPENDS "${D}/libcommon/dpcommon/*.c" "${D}/libcommon/dpcommon/stable_sort.cpp")
list(FILTER common EXCLUDE REGEX "threading_win32|threading_qt")
file(GLOB engine CONFIGURE_DEPENDS "${D}/libengine/dpengine/*.c")
file(GLOB msg CONFIGURE_DEPENDS "${D}/libmsg/dpmsg/*.c")
file(GLOB mypaint CONFIGURE_DEPENDS "${D}/bundled/libmypaint/*.c")
file(GLOB zstd CONFIGURE_DEPENDS "${Z}/common/*.c" "${Z}/compress/*.c" "${Z}/decompress/*.c")
add_library(we_drawdance_core STATIC ${common} ${engine} ${msg} ${mypaint} ${zstd}
 "${D}/libdb/dpdb/sql.c" "${D}/bundled/sqlite3/sqlite3.c" "${D}/bundled/sqlite3/dbdata.c" "${D}/bundled/sqlite3/sqlite3recover.c"
 "${D}/bundled/parson/parson.c" "${D}/bundled/qgrayraster/qgrayraster.c")
target_include_directories(we_drawdance_core PUBLIC "${D}/libcommon" "${D}/libengine" "${D}/libmsg" "${D}/libdb"
 "${D}/bundled/libmypaint" "${D}/bundled/uthash" "${D}/bundled/parson" "${D}/bundled/qgrayraster" "${D}/bundled/sqlite3" "${Z}")
target_compile_definitions(we_drawdance_core PRIVATE SQLITE_OMIT_LOAD_EXTENSION=1 SQLITE_ENABLE_DBPAGE_VTAB=1 SQLITE_THREADSAFE=2)
target_compile_options(we_drawdance_core PRIVATE -O2 -ffunction-sections -fdata-sections -fno-strict-aliasing)
add_library(we_drawpile SHARED drawpile_classic.c drawdance_adapter.c drawdance_jni.c)
target_compile_options(we_drawpile PRIVATE -O2)
target_link_libraries(we_drawpile we_drawdance_core z m)
target_link_options(we_drawpile PRIVATE -Wl,--gc-sections)
